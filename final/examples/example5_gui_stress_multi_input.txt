# Example 5: GUI stress test with multi-input gates
# This circuit is deliberately wider and denser than a teaching example.
# It checks multi-input gate parsing, crowded circuit drawing, long labels,
# many monitor traces, clocks, and DTYPE outputs.

DEVICES {
    CLK_FAST : CLOCK(2);
    CLK_SLOW : CLOCK(5);

    SW_A : SWITCH(0);
    SW_B : SWITCH(1);
    SW_C : SWITCH(0);
    SW_D : SWITCH(1);
    SW_E : SWITCH(0);
    SW_F : SWITCH(1);
    SW_G : SWITCH(0);
    SW_H : SWITCH(1);
    ENABLE_MAIN : SWITCH(1);
    RESET_MAIN : SWITCH(0);

    WIDE_AND_SIX_INPUTS : AND(6);
    WIDE_OR_FIVE_INPUTS : OR(5);
    WIDE_NAND_FIVE_INPUTS : NAND(5);
    WIDE_NOR_FIVE_INPUTS : NOR(5);

    XOR_STAGE_LEFT : XOR;
    XOR_STAGE_RIGHT : XOR;
    COMBINE_AND : AND(2);
    FINAL_OR_OUTPUT : OR(2);

    MEMORY_CELL_LONG_NAME : DTYPE;
}

CONNECT {
    # Six-input AND gate stresses parser ports I1 to I6 and GUI pin spacing.
    SW_A -> WIDE_AND_SIX_INPUTS.I1;
    SW_B -> WIDE_AND_SIX_INPUTS.I2;
    SW_C -> WIDE_AND_SIX_INPUTS.I3;
    SW_D -> WIDE_AND_SIX_INPUTS.I4;
    SW_E -> WIDE_AND_SIX_INPUTS.I5;
    ENABLE_MAIN -> WIDE_AND_SIX_INPUTS.I6;

    # Five-input OR/NAND/NOR gates give the overview several crowded blocks.
    SW_A -> WIDE_OR_FIVE_INPUTS.I1;
    SW_C -> WIDE_OR_FIVE_INPUTS.I2;
    SW_E -> WIDE_OR_FIVE_INPUTS.I3;
    SW_G -> WIDE_OR_FIVE_INPUTS.I4;
    RESET_MAIN -> WIDE_OR_FIVE_INPUTS.I5;

    SW_B -> WIDE_NAND_FIVE_INPUTS.I1;
    SW_D -> WIDE_NAND_FIVE_INPUTS.I2;
    SW_F -> WIDE_NAND_FIVE_INPUTS.I3;
    SW_H -> WIDE_NAND_FIVE_INPUTS.I4;
    ENABLE_MAIN -> WIDE_NAND_FIVE_INPUTS.I5;

    WIDE_AND_SIX_INPUTS -> WIDE_NOR_FIVE_INPUTS.I1;
    WIDE_OR_FIVE_INPUTS -> WIDE_NOR_FIVE_INPUTS.I2;
    WIDE_NAND_FIVE_INPUTS -> WIDE_NOR_FIVE_INPUTS.I3;
    CLK_SLOW -> WIDE_NOR_FIVE_INPUTS.I4;
    RESET_MAIN -> WIDE_NOR_FIVE_INPUTS.I5;

    WIDE_AND_SIX_INPUTS -> XOR_STAGE_LEFT.I1;
    WIDE_OR_FIVE_INPUTS -> XOR_STAGE_LEFT.I2;

    WIDE_NAND_FIVE_INPUTS -> XOR_STAGE_RIGHT.I1;
    WIDE_NOR_FIVE_INPUTS -> XOR_STAGE_RIGHT.I2;

    XOR_STAGE_LEFT -> COMBINE_AND.I1;
    XOR_STAGE_RIGHT -> COMBINE_AND.I2;

    COMBINE_AND -> FINAL_OR_OUTPUT.I1;
    WIDE_NOR_FIVE_INPUTS -> FINAL_OR_OUTPUT.I2;

    FINAL_OR_OUTPUT -> MEMORY_CELL_LONG_NAME.DATA;
    CLK_FAST -> MEMORY_CELL_LONG_NAME.CLK;
    RESET_MAIN -> MEMORY_CELL_LONG_NAME.SET;
    SW_G -> MEMORY_CELL_LONG_NAME.CLEAR;
}

MONITOR {
    CLK_FAST,
    CLK_SLOW,
    SW_A,
    SW_B,
    SW_C,
    SW_D,
    SW_E,
    SW_F,
    SW_G,
    SW_H,
    ENABLE_MAIN,
    RESET_MAIN,
    WIDE_AND_SIX_INPUTS,
    WIDE_OR_FIVE_INPUTS,
    WIDE_NAND_FIVE_INPUTS,
    WIDE_NOR_FIVE_INPUTS,
    XOR_STAGE_LEFT,
    XOR_STAGE_RIGHT,
    COMBINE_AND,
    FINAL_OR_OUTPUT,
    MEMORY_CELL_LONG_NAME.Q,
    MEMORY_CELL_LONG_NAME.QBAR
};

END;
