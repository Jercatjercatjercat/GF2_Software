  600:     def estimate_circuit_height(self):
  601:         """Return enough circuit height to avoid stacked device overlap."""
  602:         layers = self.calculate_device_layers()
  603:         layer_counts = {}
  604:         for device in self.devices.devices_list:
  605:             layer = layers.get(device.device_id, 1)
  606:             layer_counts[layer] = layer_counts.get(layer, 0) + 1
  607: 
  608:         max_devices_in_layer = max(layer_counts.values(), default=1)
  609:         block_height = 34
  610:         row_gap = 8
  611:         vertical_padding = 56
  612:         return (
  613:             vertical_padding
  614:             + max_devices_in_layer * block_height
  615:             + max(0, max_devices_in_layer - 1) * row_gap
  616:         )

  618:     def calculate_device_layers(self):
  619:         """Group devices by dependency depth for circuit drawing."""
  620:         layers = {}
  621:         for device in self.devices.devices_list:
  622:             if not device.inputs:
  623:                 layers[device.device_id] = 0
  624: 
  625:         for _ in self.devices.devices_list:
  626:             changed = False
  627:             for device in self.devices.devices_list:
  628:                 if device.device_id in layers:
  629:                     continue
  630: 
  631:                 source_layers = []
  632:                 waiting_for_source = False
  633:                 for connected_output in device.inputs.values():
  634:                     if connected_output is None:
  635:                         continue
  636:                     source_device_id, _ = connected_output
  637:                     if source_device_id == device.device_id:
  638:                         continue
  639:                     if source_device_id not in layers:
  640:                         waiting_for_source = True
  641:                         break
  642:                     source_layers.append(layers[source_device_id])
  643: 
  644:                 if not waiting_for_source and source_layers:
  645:                     layers[device.device_id] = max(source_layers) + 1
  646:                     changed = True
  647: 
  648:             if not changed:
  649:                 break
  650: 
  651:         for device in self.devices.devices_list:
  652:             if device.device_id not in layers:
  653:                 layers[device.device_id] = 1
  654: 
  655:         return layers

   20:     def run_from_start(self, cycles):
   21:         """Cold-start the network and run for the requested cycles."""
   22:         cycles = self.validate_cycles(cycles)
   23:         if cycles is None:
   24:             return False, "Number of cycles must be zero or greater."
   25: 
   26:         self.cycles_completed = 0
   27:         self.monitors.reset_monitors()
   28:         self.devices.cold_startup()

   30:         success, message, cycles_run = self.run_network(cycles)
   31:         self.cycles_completed += cycles_run
   32:         if success:
   33:             return True, "Running for " + str(cycles) + " cycles."
   34:         return False, message

   37:         """Continue a network that has already been run."""
   38:         cycles = self.validate_cycles(cycles)
   39:         if cycles is None:
   40:             return False, "Number of cycles must be zero or greater."
   41:         if self.cycles_completed == 0:
   42:             return False, "Nothing to continue. Run first."
   43: 
   44:         success, message, cycles_run = self.run_network(cycles)
   45:         self.cycles_completed += cycles_run
   46:         if success:
   47:             return (
   48:                 True,
   49:                 "Continuing for "
   50:                 + str(cycles)
   51:                 + " cycles. Total: "
   52:                 + str(self.cycles_completed)
   53:                 + "."
   54:             )
   55:         return False, message

   58:         """Run exactly one cycle, cold-starting if necessary."""
   59:         if self.cycles_completed == 0:
   60:             return self.run_from_start(1)
   61: 
   62:         success, message, cycles_run = self.run_network(1)
   63:         self.cycles_completed += cycles_run
   64:         if success:
   65:             return (
   66:                 True,
   67:                 "Advanced one cycle. Total: "
   68:                 + str(self.cycles_completed)
   69:                 + "."
   70:             )
   71:         return False, message

   75: 
   76:         Return a tuple containing success, message, and the number of cycles
   77:         completed before any error.
   78:         """
   79:         cycles_run = 0
   80:         for _ in range(cycles):

   82:                 return False, "Error! Network oscillating.", cycles_run

    6: class MyNames:
    7:     """Map name strings to unique integer IDs."""
    8: 
    9:     def __init__(self):
   10:         """Initialise an empty name table."""
   11:         self.names = []
   12: 
   13:     def lookup(self, name_string):
   14:         """Return the ID for name_string, adding it if necessary."""
   15:         if name_string in self.names:
   16:             return self.names.index(name_string)
   17: 
   18:         self.names.append(name_string)
   19:         return len(self.names) - 1
   20: 
   21:     def get_string(self, name_id):
   22:         """Return the string for name_id, or None if the ID is invalid."""
   23:         if isinstance(name_id, int) and 0 <= name_id < len(self.names):
   24:             return self.names[name_id]
   25: 
   26:         return None
   27: 
   28: 
   29: def open_file(path):
   30:     """Open the file at path for reading and return the file object."""
   31:     try:
   32:         return open(path, "r", encoding="utf-8")
